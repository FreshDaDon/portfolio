01 / Access control and filtered views Python / Django +

Authenticated users retrieve their own bookings; group membership restricts management access. 

from django.contrib.auth.decorators import login_required, user_passes_test
from django.db.models import Q
from django.shortcuts import render


def is_manager(user):
    return (
        user.is_authenticated
        and user.groups.filter(name="Managers").exists()
    )


@login_required
def my_requests(request):
    bookings = (
        BookingRequest.objects
        .filter(user=request.user)
        .order_by("-date_of_use")
    )

    return render(
        request,
        "studio/booking/my_requests.html",
        {"bookings": bookings},
    )


@login_required
@user_passes_test(is_manager)
def view_requests(request):
    query = request.GET.get("q", "").strip()

    filters = (
        Q(reviewed=False)
        & ~Q(status="Cancelled")
        & ~Q(status="Rejected")
    )

    if query:
        filters &= (
            Q(request_number__icontains=query)
            | Q(staff_number__icontains=query)
        )

    booking_requests = (
        BookingRequest.objects
        .filter(filters)
        .order_by("-date_of_use")
    )

    context = {
        "requests": booking_requests,
        "search_query": query,
    }

    return render(
        request,
        "studio/manager/view_requests.html",
        context,
    )
02 / Booking validation and submission Python / Django +

Server-side rules check dates, duration, operating hours and overlapping reservations before saving. 

from datetime import datetime, time, timedelta

from django.contrib.auth.decorators import login_required
from django.db import transaction
from django.shortcuts import redirect, render
from django.utils import timezone


def validate_booking_slot(booking):
    if booking.time_start >= booking.time_end:
        return "Start time must be before end time."

    start_dt = datetime.combine(
        booking.date_of_use,
        booking.time_start,
    )
    end_dt = datetime.combine(
        booking.date_of_use,
        booking.time_end,
    )

    if end_dt - start_dt > timedelta(hours=1):
        return "Bookings longer than one hour are not allowed."

    if booking.date_of_use < timezone.localdate():
        return "The booking date cannot be in the past."

    if booking.date_of_use.weekday() >= 5:
        return "Studios are unavailable on weekends."

    closing_time = time(16, 30)

    if (
        booking.time_start > closing_time
        or booking.time_end > closing_time
    ):
        return "Bookings may not extend beyond 16:30."

    conflicting_booking = (
        BookingRequest.objects.filter(
            venue_required=booking.venue_required,
            date_of_use=booking.date_of_use,
            time_start__lt=booking.time_end,
            time_end__gt=booking.time_start,
        )
        .exclude(
            status__in=[
                StatusChoices.CANCELLED,
                StatusChoices.REJECTED,
            ]
        )
        .exclude(pk=booking.pk)
        .exists()
    )

    if conflicting_booking:
        return "This venue is already booked for the selected time slot."

    return None


@login_required
@transaction.atomic
def make_booking(request):
    if request.method == "POST":
        form = BookingRequestForm(request.POST)

        if form.is_valid():
            booking = form.save(commit=False)
            booking.user = request.user

            validation_error = validate_booking_slot(booking)

            context = {
                "form": form,
                "error_message": validation_error,
            }

            if validation_error:
                return render(
                    request,
                    "studio/booking/make_booking.html",
                    context,
                )

            booking.status = determine_initial_status(request.user)
            booking.save()
            form.save_m2m()

            send_booking_notification(booking)

            return redirect(
                "request_detail",
                request_number=booking.request_number,
            )
    else:
        form = BookingRequestForm()

    return render(
        request,
        "studio/booking/make_booking.html",
        {"form": form},
    )
03 / Moodle monthly assessment activity MariaDB · Moodle reporting

This anonymised excerpt identifies the number of Moodle assignments and quizzes created in each module during a defined reporting month. I used this type of query when extracting data for monthly faculty reports and Power BI dashboards. 

SELECT
    c.shortname AS module_code,
    COUNT(DISTINCT cm.id) AS assessments_created
FROM mdl_course AS c
JOIN mdl_course_modules AS cm
    ON cm.course = c.id
JOIN mdl_modules AS m
    ON m.id = cm.module
WHERE m.name IN ('assign', 'quiz')
  AND FROM_UNIXTIME(cm.added) >= '2026-09-01'
  AND FROM_UNIXTIME(cm.added) <  '2026-10-01'
GROUP BY c.shortname
ORDER BY assessments_created DESC;

Purpose

The query measures assessment-development activity within a defined monthly reporting period. The exclusive upper boundary prevents activities created during the following month from being included.

Design decisions

Moodle stores activity creation dates as Unix timestamps. MariaDB’s FROM_UNIXTIME() function converts each value into a date and time that can be filtered. The query joins mdl_modules to include only assignments and quizzes, while COUNT(DISTINCT cm.id) prevents duplicate activity records from inflating the totals.

Security and privacy

The production query was executed using read-only reporting access. This anonymised excerpt returns aggregated module-level information and excludes student names, student numbers, grades and other personal data.

Reporting application

The extracted results were transformed and incorporated into Power BI to support monthly faculty-level analysis of Moodle assessment activity.
